`timescale 1ns/1ps

module tb_Industrial_Machine_Health;

    reg [3:0] sensor_value;

    wire normal_led;
    wire warning_led;
    wire critical_led;
    wire alarm;

    Industrial_Machine_Health uut (
        .sensor_value(sensor_value),
        .normal_led(normal_led),
        .warning_led(warning_led),
        .critical_led(critical_led),
        .alarm(alarm)
    );

    initial begin

        // Normal condition
        sensor_value = 4'd0;
        #10;

        // Normal condition - upper limit
        sensor_value = 4'd7;
        #10;

        // Warning condition - lower limit
        sensor_value = 4'd8;
        #10;

        // Warning condition
        sensor_value = 4'd10;
        #10;

        // Warning condition - upper limit
        sensor_value = 4'd11;
        #10;

        // Critical condition
        sensor_value = 4'd12;
        #10;

        // Critical condition - maximum value
        sensor_value = 4'd15;
        #10;

        $stop;

    end

endmodule
